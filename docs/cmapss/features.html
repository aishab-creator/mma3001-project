cmapss.features

Leakage-safe feature engineering.

Two kinds of feature are built:

  • Temporal features (add_temporal_features()): rolling mean and standard deviation, first difference and (optionally) trend features. Every one is computed per engine with a trailing window, so the value at cycle t depends only on cycles <= t of the same engine. No future sensor value can leak into a feature, and one engine cannot influence another.
  • Operating-regime features (add_regimes()): operating settings are clustered into regimes and each sensor is standardised within its regime. The clustering and the regime statistics are learned on the fit partition only and then applied to the other partition.

 30def add_temporal_features(df, sensors, window=5, add_trends=False):
 31    """Append rolling and difference features for each sensor.
 32
 33    Parameters
 34    ----------
 35    df : pandas.DataFrame
 36        Table with ``engine_id``, ``cycle`` and the ``sensors`` columns.
 37    sensors : list of str
 38        Sensor columns to expand.
 39    window : int, optional
 40        Trailing window length in cycles (default 5).
 41    add_trends : bool, optional
 42        If True, also add ``<sensor>_slope`` and ``<sensor>_change_from_start``.
 43
 44    Returns
 45    -------
 46    pandas.DataFrame
 47        Copy of ``df`` (sorted by engine and cycle) with new columns
 48        ``<sensor>_rolling_mean``, ``_rolling_std``, ``_delta`` and, if
 49        requested, ``_slope`` and ``_change_from_start``.
 50
 51    Notes
 52    -----
 53    * ``rolling_std`` is undefined for the first cycle of an engine (one
 54      observation); it is filled with 0 there. ``delta`` is also 0 at the
 55      first cycle. Neither value is imputed from other engines.
 56    * ``slope`` is the change over the last ``window - 1`` cycles divided by
 57      the number of cycles actually available (at least 1), so early cycles
 58      use a shorter span instead of reading data from before the engine began.
 59    * ``change_from_start`` is the sensor minus that engine's first reading.
 60    * All windows look backwards only, which is what prevents leakage.
 61    """
 62    df = df.sort_values(["engine_id", "cycle"]).copy()
 63    group = df.groupby("engine_id", sort=False)
 64    new_cols = {}
 65    for s in sensors:
 66        # min_periods=1 lets the window shrink at the start of an engine.
 67        new_cols[f"{s}_rolling_mean"] = group[s].transform(
 68            lambda x: x.rolling(window, min_periods=1).mean()
 69        )
 70        new_cols[f"{s}_rolling_std"] = group[s].transform(
 71            lambda x: x.rolling(window, min_periods=1).std()
 72        ).fillna(0)
 73        new_cols[f"{s}_delta"] = group[s].diff().fillna(0)
 74        if add_trends:
 75            new_cols[f"{s}_slope"] = group[s].transform(
 76                lambda x: (x - x.shift(window - 1))
 77                / np.minimum(np.arange(len(x)), window - 1).clip(min=1)
 78            ).fillna(0)
 79            new_cols[f"{s}_change_from_start"] = df[s] - group[s].transform("first")
 80    return pd.concat([df, pd.DataFrame(new_cols, index=df.index)], axis=1)

 83def temporal_columns(sensors, add_trends=False):
 84    """List the column names that :func:`add_temporal_features` creates.
 85
 86    Parameters
 87    ----------
 88    sensors : list of str
 89        Sensor columns that were expanded.
 90    add_trends : bool, optional
 91        Whether trend features were requested.
 92
 93    Returns
 94    -------
 95    list of str
 96        Names in sensor-major order, e.g. ``sensor_2_rolling_mean``.
 97    """
 98    suffixes = ["rolling_mean", "rolling_std", "delta"]
 99    if add_trends:
100        suffixes += ["slope", "change_from_start"]
101    return [f"{s}_{suffix}" for s in sensors for suffix in suffixes]

104def add_regimes(fit_df, other_df, sensors, n_regimes=6, random_state=42):
105    """Add operating-regime id and regime-standardised sensors.
106
107    Settings are scaled and clustered with k-means **on** ``fit_df`` **only**;
108    ``other_df`` is assigned to the nearest learned cluster. For each sensor,
109    the per-regime mean and standard deviation are likewise taken from
110    ``fit_df`` and applied to both frames. This removes the sensor swing
111    caused by throttle/load changes (FD002, FD004 have six conditions).
112
113    Parameters
114    ----------
115    fit_df, other_df : pandas.DataFrame
116        Training-side and evaluation-side tables. ``other_df`` never
117        influences the clusters or statistics.
118    sensors : list of str
119        Sensors to standardise.
120    n_regimes : int, optional
121        Number of k-means clusters (default 6, the number of C-MAPSS
122        operating conditions in FD002/FD004).
123    random_state : int, optional
124        Seed for k-means.
125
126    Returns
127    -------
128    (pandas.DataFrame, pandas.DataFrame)
129        Copies of ``fit_df`` and ``other_df`` with ``operating_regime`` and
130        ``<sensor>_regime_z`` columns.
131    """
132    scaler = StandardScaler().fit(fit_df[SETTING_COLS])
133    fit_settings = scaler.transform(fit_df[SETTING_COLS])
134    other_settings = scaler.transform(other_df[SETTING_COLS])
135    k = min(n_regimes, len(fit_df))
136    kmeans = KMeans(n_clusters=k, n_init=10, random_state=random_state).fit(
137        fit_settings
138    )
139    fit = fit_df.copy()
140    other = other_df.copy()
141    fit["operating_regime"] = kmeans.predict(fit_settings)
142    other["operating_regime"] = kmeans.predict(other_settings)
143    for s in sensors:
144        # Regime statistics come from the fit partition alone.
145        stats = fit.groupby("operating_regime")[s].agg(["mean", "std"])
146        overall_std = float(fit[s].std())
147        if not np.isfinite(overall_std) or overall_std < MIN_SENSOR_STD:
148            overall_std = 1.0  # avoid dividing by ~0 for a flat sensor
149        means_fit = fit["operating_regime"].map(stats["mean"])
150        means_other = other["operating_regime"].map(stats["mean"])
151        # A regime with a single row has NaN std: fall back to the overall std.
152        stds_fit = (
153            fit["operating_regime"].map(stats["std"]).fillna(overall_std).clip(lower=MIN_SENSOR_STD)
154        )
155        stds_other = (
156            other["operating_regime"].map(stats["std"]).fillna(overall_std).clip(lower=MIN_SENSOR_STD)
157        )
158        fit[f"{s}_regime_z"] = (fit[s] - means_fit) / stds_fit
159        other[f"{s}_regime_z"] = (other[s] - means_other) / stds_other
160    return fit, other

163def build_features(fit_raw, other_raw, variant, window=5, random_state=42):
164    """Build the feature table for one model variant.
165
166    Parameters
167    ----------
168    fit_raw : pandas.DataFrame
169        Validated training-side table (output of
170        :func:`cmapss.data.prepare_frame`).
171    other_raw : pandas.DataFrame
172        Validated evaluation-side table (validation snapshots or test data).
173    variant : {"Improved RF", "RF + trends", "RF + regimes"}
174        ``"Improved RF"`` uses rolling mean/std and deltas;
175        ``"RF + trends"`` adds slope and change-from-start;
176        ``"RF + regimes"`` adds operating-regime features.
177    window : int, optional
178        Rolling window in cycles.
179    random_state : int, optional
180        Seed for the regime k-means.
181
182    Returns
183    -------
184    (pandas.DataFrame, pandas.DataFrame, list of str)
185        Feature tables for the fit and other partitions, and the ordered list
186        of model feature column names.
187
188    Raises
189    ------
190    ValueError
191        If ``variant`` is not recognised.
192
193    Notes
194    -----
195    Near-constant sensors (std <= 1e-8) are dropped. That decision uses
196    ``fit_raw`` only, so the evaluation data cannot influence which inputs
197    the model sees.
198    """
199    if variant not in VARIANTS:
200        raise ValueError(f"unknown variant {variant!r}; choose from {VARIANTS}")
201    sensors = [s for s in SENSOR_COLS if fit_raw[s].std() > MIN_SENSOR_STD]
202    add_trends = variant == "RF + trends"
203    fit = add_temporal_features(fit_raw, sensors, window, add_trends)
204    other = add_temporal_features(other_raw, sensors, window, add_trends)
205    features = (
206        ["cycle"] + SETTING_COLS + sensors + temporal_columns(sensors, add_trends)
207    )
208    if variant == "RF + regimes":
209        fit, other = add_regimes(fit, other, sensors, random_state=random_state)
210        features += ["operating_regime"] + [f"{s}_regime_z" for s in sensors]
211    return fit, other, features
VARIANTS = ('Improved RF', 'RF + trends', 'RF + regimes')
MIN_SENSOR_STD = 1e-08
